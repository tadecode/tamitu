import 'package:flutter/rendering.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:material_ui/material_ui.dart';
import 'package:tamitu/core/errors/app_failure.dart';
import 'package:tamitu/core/errors/failure_messages.dart';
import 'package:tamitu/core/ui/app_icon.dart';
import 'package:tamitu/core/ui/components.dart';
import 'package:tamitu/l10n/app_localizations.dart';

import '../helpers.dart';

class _Host extends StatelessWidget {
  const _Host(this.failure);
  final AppFailure failure;
  @override
  Widget build(BuildContext context) {
    final m = failureMessage(AppLocalizations.of(context)!, failure,
        formatTime: (t) => '00:00');
    return Scaffold(
      appBar: AppBar(title: const Text('Tamitu')),
      body: StateView(
        icon: AppIcons.error,
        title: m.title ?? '',
        body: m.body,
        primary: FilledButton(onPressed: () {}, child: const Text('Retry')),
        secondary: TextButton(onPressed: () {}, child: const Text('Back')),
      ),
    );
  }
}

void main() {
  const sizes = [Size(320, 568), Size(360, 640)];
  const scales = [1.0, 1.3, 2.0];
  for (final lang in languages) {
    for (final code in ['ai_unavailable', 'ai_invalid_response']) {
      for (final slot in [true, false, null]) {
        for (final size in sizes) {
          for (final scale in scales) {
            testWidgets('$lang $code slot=$slot ${size.width.toInt()}x${size.height.toInt()} x$scale', (tester) async {
              tester.view.physicalSize = size * 3;
              tester.view.devicePixelRatio = 3;
              tester.platformDispatcher.textScaleFactorTestValue = scale;
              addTearDown(tester.view.reset);
              addTearDown(tester.platformDispatcher.clearAllTestValues);
              final f = classifyCallableError(code: 'internal', details: {
                'code': code,
                if (slot != null) 'slotReleased': slot,
              });
              await tester.pumpWidget(hostScreen(TestEnv(), _Host(f), locale: lang));
              await tester.pumpAndSettle();
              expect(tester.takeException(), isNull);
              final l10n = await AppLocalizations.delegate.load(Locale(lang));
              final body = failureMessage(l10n, f, formatTime: (t) => '').body;
              final finder = find.text(body);
              expect(finder, findsOneWidget);
              final p = tester.renderObject<RenderParagraph>(finder);
              expect(p.didExceedMaxLines, isFalse);
              expect(p.textSize.height, lessThanOrEqualTo(p.size.height + 0.5));
              await tester.ensureVisible(finder);
              await tester.pumpAndSettle();
              final r0 = tester.getRect(finder);
              expect(r0.left >= 0 && r0.right <= size.width, isTrue, reason: '$r0');
              final tall = r0.height > size.height - 56;
              final pos = tester
                  .state<ScrollableState>(find.byType(Scrollable).last)
                  .position;
              pos.jumpTo(pos.maxScrollExtent);
              await tester.pumpAndSettle();
              final r = tester.getRect(finder);
              expect(r.bottom <= size.height, isTrue, reason: '$r');
              expect(tester.getRect(find.text('Back')).bottom <= size.height, isTrue);
              // ignore: avoid_print
              print('FIT $lang $code $slot ${size.width.toInt()}x${size.height.toInt()} x$scale h=${r0.height} tall=$tall');
            });
          }
        }
      }
    }
  }
}
